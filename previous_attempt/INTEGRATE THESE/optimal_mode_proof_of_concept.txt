## 1. Feature Tracking Core (src/android/compile_time/feature_tracker.nim)

type
  AndroidFeature* = enum
    # Core features
    afLogging, afJNI
    # Audio features  
    afAudio, afLowLatencyAudio, afAudioEffects
    # Graphics features
    afGraphics, afOpenGLES2, afOpenGLES3, afVulkan
    # Input features
    afInput, afSensors, afGameController
    # System features
    afNotifications, afVibration, afCamera
    # Network features
    afNetwork, afBluetooth, afWiFi

  FeatureInfo* = object
    minApiLevel*: int
    deviceCoverage*: float  # Percentage of devices that support this
    needsJavaContext*: bool
    fallbackAvailable*: bool
    fallbackFeature*: AndroidFeature
    description*: string

# Global compile-time tracking
var 
  usedFeatures* {.compileTime.}: set[AndroidFeature]
  needsJavaSDK* {.compileTime.}: bool = false
  detectedMinApi* {.compileTime.}: int = 21

const FeatureDatabase* = {
  afLogging: FeatureInfo(
    minApiLevel: 21, deviceCoverage: 100.0, needsJavaContext: false,
    description: "Android logging system"),
  afAudio: FeatureInfo(
    minApiLevel: 21, deviceCoverage: 100.0, needsJavaContext: false,
    fallbackAvailable: true, description: "Basic audio playback"),
  afLowLatencyAudio: FeatureInfo(
    minApiLevel: 26, deviceCoverage: 85.0, needsJavaContext: false,
    fallbackAvailable: true, fallbackFeature: afAudio,
    description: "Professional low-latency audio (AAudio)"),
  afVulkan: FeatureInfo(
    minApiLevel: 24, deviceCoverage: 75.0, needsJavaContext: false,
    fallbackAvailable: true, fallbackFeature: afOpenGLES3,
    description: "High-performance Vulkan graphics"),
  afNotifications: FeatureInfo(
    minApiLevel: 21, deviceCoverage: 100.0, needsJavaContext: true,
    description: "System notifications"),
  afCamera: FeatureInfo(
    minApiLevel: 24, deviceCoverage: 95.0, needsJavaContext: false,
    description: "Camera access via NDK")
}.toTable

template trackFeature*(feature: AndroidFeature) =
  static:
    usedFeatures.incl(feature)
    let info = FeatureDatabase[feature]
    detectedMinApi = max(detectedMinApi, info.minApiLevel)
    if info.needsJavaContext:
      needsJavaSDK = true

macro analyzeAndroidFeatures*(): untyped =
  var output = newSeq[string]()
  
  output.add("\n🔍 Android Feature Analysis:")
  output.add("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
  
  # Calculate minimum API and coverage
  var minCoverage = 100.0
  for feature in usedFeatures:
    let info = FeatureDatabase[feature]
    minCoverage = min(minCoverage, info.deviceCoverage)
  
  output.add("📱 Minimum API Level: " & $detectedMinApi)
  output.add("📊 Device Coverage: ~" & $(minCoverage.int) & "%")
  
  # SDK requirement analysis
  if needsJavaSDK:
    output.add("☕ Java SDK Required: YES")  
    output.add("📦 Add to build.gradle:")
    output.add("   implementation 'org.nim-lang:android-runtime:1.0.0'")
    output.add("🔧 Required permissions will be auto-generated")
  else:
    output.add("🚀 Pure NDK: YES (maximum performance)")
    output.add("📦 No additional dependencies required")
  
  # Feature-specific recommendations
  output.add("\n💡 Feature Analysis:")
  for feature in usedFeatures:
    let info = FeatureDatabase[feature]
    output.add("   • " & info.description)
    if info.minApiLevel > 21:
      output.add("     ⚠️  Requires API " & $info.minApiLevel & "+ (" & $(info.deviceCoverage.int) & "% coverage)")
    if info.fallbackAvailable:
      let fallback = FeatureDatabase[info.fallbackFeature]
      output.add("     🔄 Fallback: " & fallback.description)
  
  # Generate the echo statements
  result = newNimNode(nnkStmtList)
  for line in output:
    result.add(newCall("echo", newLit(line)))

## 4. Runtime Implementation Selection

type
  AndroidRuntime* = ref object
    mode*: AndroidRuntimeMode
    apiLevel*: int32
    context*: Option[AndroidContext]
    availableFeatures*: set[AndroidFeature]

  AndroidRuntimeMode* = enum
    armAuto        # Automatic best choice (default)
    armNdkOnly     # Pure NDK performance
    armSdkOnly     # Pure SDK features
    armHybrid      # Best of both worlds

proc initAndroidRuntime*(mode = armAuto): AndroidRuntime =
  new(result)
  result.apiLevel = getCurrentApiLevel()
  
  # Determine optimal mode based on usage
  let optimalMode = static:
    if needsJavaSDK and (afLowLatencyAudio in usedFeatures or afVulkan in usedFeatures):
      armHybrid  # Need both performance and SDK features
    elif needsJavaSDK:
      armSdkOnly # SDK features sufficient
    else:
      armNdkOnly # Pure performance
  
  result.mode = if mode == armAuto: optimalMode else: mode
  
  # Set up context based on mode
  case result.mode
  of armNdkOnly:
    result.context = none(AndroidContext)
  of armSdkOnly, armHybrid, armAuto:
    result.context = autoDetectContext()
  
  result.availableFeatures = detectAvailableFeatures(result)

# This would be automatically generated based on feature analysis
when defined(android):
  static:
    # Generate Android manifest permissions
    var permissions: seq[string] = @[]
    if afCamera in usedFeatures:
      permissions.add("CAMERA")
    if afAudio in usedFeatures:
      permissions.add("RECORD_AUDIO")
    if afVibration in usedFeatures:
      permissions.add("VIBRATE")
    if afNetwork in usedFeatures:
      permissions.add("INTERNET")
    if afBluetooth in usedFeatures:
      permissions.add("BLUETOOTH")
      permissions.add("BLUETOOTH_ADMIN")
    
    # Generate linker flags based on used features
    var linkFlags: seq[string] = @["-landroid", "-llog"]
    if afAudio in usedFeatures or afLowLatencyAudio in usedFeatures:
      linkFlags.add("-laaudio")
      linkFlags.add("-lOpenSLES")
    if afGraphics in usedFeatures or afVulkan in usedFeatures:
      linkFlags.add("-lEGL")
      linkFlags.add("-lGLESv2")
      linkFlags.add("-lGLESv3")
      linkFlags.add("-lvulkan")
    if afCamera in usedFeatures:
      linkFlags.add("-lcamera2ndk")
      linkFlags.add("-lmediandk")
    
    # Apply the configuration
    for flag in linkFlags:
      {.passL: flag.}
    
    # Generate Android manifest template
    writeFile("android_manifest_template.xml", generateManifest(permissions))
    
    # Generate Gradle dependencies
    if needsJavaSDK:
      writeFile("gradle_dependencies.txt", generateGradleDeps())

proc generateManifest(permissions: seq[string]): string =
  result = """<?xml version="1.0" encoding="utf-8"?>
<manifest xmlns:android="http://schemas.android.com/apk/res/android">
"""
  for perm in permissions:
    result.add("    <uses-permission android:name=\"android.permission." & perm & "\" />\n")
  
  result.add("""    
    <application android:label="NimAndroidApp">
        <activity android:name=".MainActivity"
                  android:exported="true">
            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <category android:name="android.intent.category.LAUNCHER" />
            </intent-filter>
        </activity>
    </application>
</manifest>""")

## 7. Smart Context Detection and Management

proc autoDetectContext*(): Option[AndroidContext] =
  ## Automatically detect Android context from various sources
  
  # Priority 1: GameActivity (best for games/interactive apps)
  when compiles(getGameActivityContext()):
    if let ctx = getGameActivityContext():
      echo "📱 Using GameActivity context (optimal for games)"
      return some(ctx)
  
  # Priority 2: NativeActivity (fallback for pure native apps)
  when compiles(getNativeActivityContext()):
    if let ctx = getNativeActivityContext():
      echo "📱 Using NativeActivity context"
      return some(ctx)
  
  # Priority 3: JNI global reference (manual setup)
  if globalJNIContext.isSome:
    echo "📱 Using manually provided JNI context"
    return globalJNIContext
  
  # Priority 4: Auto-generate setup instructions
  when needsJavaSDK:
    {.error: """
🚫 Java SDK context required but not available!

Your app uses features that need Java SDK access:""" & 
    (when afNotifications in usedFeatures: "\n  • Notifications" else: "") &
    (when afVibration in usedFeatures: "\n  • Vibration" else: "") &
    (when afCamera in usedFeatures: "\n  • Camera" else: "") &
"""

Add ONE of these to your android_main() function:

Option 1 - GameActivity (Recommended for games):
  proc android_main(app: ptr android_app) =
    setGameActivityContext(app)
    yourMainFunction()

Option 2 - Manual JNI setup:
  proc android_main(app: ptr android_app) =
    let context = AndroidContext(
      jniEnv: app.activity.env,
      javaActivity: app.activity.clazz
    )
    setAndroidContext(context)
    yourMainFunction()

Option 3 - Pure NDK mode (remove SDK features):
  # Remove calls to: showNotification(), vibrate(), etc.
  # Or use NDK alternatives where available
""".}
  
  return none(AndroidContext)

proc generateNimbleFile*(projectName: string): string =
  let minApi = detectedMinApi
  let features = usedFeatures
  let needsSDK = needsJavaSDK
  
  result = """# Package
version       = "1.0.0"
author        = "Your Name"
description   = "Android app built with Nim"
license       = "MIT"
bin           = @["$1"]

# Dependencies  
requires "nim >= 1.6.0"
requires "android >= 1.0.0"
""" % [projectName]

  if needsSDK:
    result.add("""
# Android-specific configuration
task android, "Build for Android":
  exec "nim c --os:linux --cpu:arm64 -d:android -d:minApiLevel:$1 src/$2.nim"
""" % [$minApi, projectName])

# CMake integration for Android Studio
proc generateCMakeFile*(): string =
  result = """cmake_minimum_required(VERSION 3.18.1)
project(NimAndroidApp)

# Find required libraries based on detected features
"""
  
  if afAudio in usedFeatures:
    result.add("find_library(aaudio-lib aaudio)\n")
    result.add("find_library(opensl-lib OpenSLES)\n")
  
  if afGraphics in usedFeatures:
    result.add("find_library(egl-lib EGL)\n")
    result.add("find_library(gles-lib GLESv3)\n")
  
  if afVulkan in usedFeatures:
    result.add("find_library(vulkan-lib vulkan)\n")
  
  result.add("""
# Link all required libraries
target_link_libraries(native-lib
                     ${log-lib}
                     ${android-lib}""")
  
  if afAudio in usedFeatures:
    result.add("\n                     ${aaudio-lib}")
    result.add("\n                     ${opensl-lib}")
  
  result.add(")")
