# Android Managers Architecture Design for Nim

## Overview

This design provides a comprehensive set of Android managers with clean, idiomatic Nim APIs. Each manager handles device capabilities detection, API version compatibility, and provides both singleton services and instance-based objects where appropriate.

## Core Architecture Principles

1. **Hardware-Aware Initialization**: Each manager detects device capabilities and chooses optimal APIs
2. **Clean Nim APIs**: Simple, intuitive interfaces like `audio.play()`, `vibrate(300.ms)`, `notify(title, msg)`
3. **Preference Management**: Both system-level and instance-level preferences
4. **Strategy Pattern**: Internal strategies for different API levels and capabilities
5. **Error Handling**: Graceful fallbacks and clear error reporting

## Manager Initialization and Strategy Selection

### Core Initialization Pattern

```nim
type
  AndroidDeviceInfo* = object
    apiLevel*: int
    hasLowLatencyAudio*: bool
    hasAmplitudeControl*: bool
    hasCamera2Api*: bool
    hasGooglePlayServices*: bool
    manufacturer*: string
    model*: string

proc detectDeviceCapabilities*(): AndroidDeviceInfo

proc newAudioManager*(deviceInfo: AndroidDeviceInfo): AudioManager =
  result = AudioManager()
  
  # Choose strategy based on device capabilities
  if deviceInfo.hasLowLatencyAudio:
    result.strategy = LowLatencyAudioStrategy()
  elif deviceInfo.apiLevel >= 21:
    result.strategy = AudioTrackStrategy()
  else:
    result.strategy = MediaPlayerStrategy()
    
  # Initialize default preferences
  result.systemPrefs = AudioSystemPrefs(
    ringerMode: rmNormal,
    systemVolume: 70,
    microphoneMuted: false,
    speakerphoneEnabled: false,
    bluetoothEnabled: false
  )

proc newVibrationManager*(deviceInfo: AndroidDeviceInfo): VibrationManager =
  result = VibrationManager()
  
  if deviceInfo.apiLevel >= 26:
    result.strategy = VibrationEffectStrategy()
  else:
    result.strategy = LegacyVibrationStrategy()
    
  result.systemPrefs = VibrationSystemPrefs(
    enabled: true,
    intensity: viMedium
  )
```

## ResurrectionManager - Making Mobile Feel Like Desktop

### Core Problem
Android can kill your app at any time due to memory pressure, battery optimization, or user navigation. Unlike desktop apps that run continuously, mobile apps must handle:
- Process death and restoration
- Activity lifecycle (pause/resume/destroy)
- Background limitations
- State loss during configuration changes
- Complex app startup scenarios

### Solution: Transparent State Management

```nim
type
  ResurrectionManager* = ref object
    runtime: AndroidRuntime
    stateStore: StateStore
    backgroundTasks: BackgroundTaskManager
    appState: AppState
    resurrectCallbacks: seq[ResurrectCallback]
    
  AppState* = enum
    asStarting,      # App launching
    asActive,        # Foreground and visible
    asPaused,        # Background but alive
    asSuspended,     # System might kill us
    asDead,          # Process killed
    asResurrecting   # Coming back from death
    
  ResurrectCallback* = proc(previousState: AppState, restoredData: JsonNode)
  
  StateStore* = ref object
    persistentData: JsonNode  # Survives process death
    sessionData: Table[string, JsonNode]  # Lost on process death
    tempData: Table[string, JsonNode]     # Lost on pause

# Desktop-like API - your app just "keeps running"
proc registerForResurrection*(manager: ResurrectionManager, 
                              key: string, 
                              saveState: proc(): JsonNode,
                              restoreState: proc(data: JsonNode))

proc keepAlive*(manager: ResurrectionManager, key: string, data: JsonNode)
proc keepTemporary*(manager: ResurrectionManager, key: string, data: JsonNode)
proc keepSession*(manager: ResurrectionManager, key: string, data: JsonNode)

# Your app just saves data normally, resurrection handles the rest
proc saveAppData*(manager: ResurrectionManager, key: string, data: JsonNode) =
  # Automatically determines persistence level needed
  # Critical data -> persistent (survives process death)
  # UI state -> session (survives backgrounding)
  # Temporary -> memory only
  
proc getAppData*(manager: ResurrectionManager, key: string): Option[JsonNode]
```

### Automatic State Categories

```nim
type
  StatePersistence* = enum
    spTemporary,   # Lost when activity pauses (cached images, temp UI state)
    spSession,     # Lost when process dies (current screen, user inputs)  
    spPersistent   # Survives everything (user data, settings, progress)
    
  StateManager* = ref object
    autoSave: bool
    autoClassify: bool  # Automatically determine persistence level
    
# Automatic classification based on data type and usage patterns
proc classifyData(key: string, data: JsonNode): StatePersistence =
  if key.startsWith("temp_") or key.startsWith("cache_"):
    return spTemporary
  elif key.startsWith("ui_") or key.startsWith("screen_"):
    return spSession  
  elif key.startsWith("user_") or key.startsWith("settings_"):
    return spPersistent
  else:
    # Smart classification based on data content
    if data.hasKey("userId") or data.hasKey("permanent"):
      return spPersistent
    elif data.hasKey("screenState") or data.hasKey("inputText"):
      return spSession
    else:
      return spTemporary

# Your code just saves data - the system handles the complexity
proc saveGameProgress*(game: GameManager, level: int, score: int) =
  let data = %*{"level": level, "score": score, "timestamp": now()}
  game.resurrection.keepAlive("game_progress", data)  # Auto-classified as persistent

proc saveCurrentUI*(ui: UIManager, screenState: JsonNode) =
  ui.resurrection.keepSession("current_screen", screenState)  # Auto-classified as session
```

### Transparent State Restoration

```nim
# Your app code looks like normal desktop code
type
  MusicApp* = object
    runtime: AndroidRuntime
    currentSong: string
    playlist: seq[string]
    position: Duration
    volume: int

proc initMusicApp*(): MusicApp =
  result.runtime = newAndroidRuntime()
  
  # Register for automatic resurrection
  result.runtime.resurrection.registerForResurrection("music_state") do() -> JsonNode:
    # Save current state
    return %*{
      "currentSong": result.currentSong,
      "playlist": result.playlist,
      "position": result.position.inMilliseconds,
      "volume": result.volume
    }
  do(data: JsonNode):
    # Restore state after resurrection
    if data != nil:
      result.currentSong = data["currentSong"].getStr()
      result.playlist = data["playlist"].getElems().mapIt(it.getStr())
      result.position = data["position"].getInt().milliseconds
      result.volume = data["volume"].getInt()
      
      # Continue playing from where we left off
      if result.currentSong != "":
        result.runtime.audio.play(result.currentSong)
        result.runtime.audio.seekTo(result.position)

proc playSong*(app: var MusicApp, song: string) =
  app.currentSong = song
  app.runtime.audio.play(song)
  
  # State automatically saved for resurrection
  app.runtime.resurrection.keepAlive("current_song", %song)

# Even if Android kills the app and user restarts it hours later,
# the music continues from where it left off - just like desktop!
```

## Thread Management & Foreground Services

### The Foreground Thread Problem
Android aggressively kills background processes to save battery. When your app goes to background, it has ~30 seconds before potential death. For long-running tasks, you need foreground services.

```nim
type
  ThreadManager* = ref object
    resurrection: ResurrectionManager
    foregroundServices: Table[string, ForegroundService]
    backgroundTasks: Table[string, BackgroundTask]
    
  ForegroundService* = ref object
    id: string
    notification: Notification  # Required for foreground services
    callback: proc()
    isRunning: bool
    
  TaskPriority* = enum
    tpBackground,     # Can be killed anytime (WorkManager)
    tpImportant,      # Deferred execution (JobScheduler) 
    tpCritical,       # Must stay alive (ForegroundService)
    tpRealtime        # Immediate execution (Main thread)

# Clean API that handles thread complexity
proc runInForeground*(manager: ThreadManager, 
                     id: string,
                     title: string,
                     task: proc()) =
  # Automatically creates foreground service with notification
  let notification = Notification(
    title: title,
    message: "Running in background",
    style: nsBasic
  )
  
  let service = ForegroundService(
    id: id,
    notification: notification,
    callback: task
  )
  
  manager.foregroundServices[id] = service
  # Platform handles the Android Service lifecycle

proc scheduleTask*(manager: ThreadManager, 
                  id: string,
                  priority: TaskPriority,
                  task: proc()) =
  case priority:
    of tpBackground:
      # Use WorkManager - can be delayed/batched
      scheduleBackgroundWork(id, task)
    of tpImportant:  
      # Use JobScheduler - respects system constraints
      scheduleJob(id, task)
    of tpCritical:
      # Use ForegroundService - guaranteed execution
      manager.runInForeground(id, "Critical Task", task)
    of tpRealtime:
      # Run on main thread immediately
      runOnMainThread(task)

# Usage examples:
proc startMusicStreaming*(app: MusicApp) =
  # This MUST stay alive even in background
  app.threadManager.runInForeground("music_stream", "Playing Music") do:
    while app.isStreaming:
      streamNextChunk()
      sleep(100.milliseconds)

proc syncUserData*(app: MyApp) =
  # This can wait for good conditions (WiFi, charging)
  app.threadManager.scheduleTask("data_sync", tpImportant) do:
    uploadUserData()
```

  if result.available:
    result.version = getGooglePlayServicesVersion()

    if hasFusedLocation(): result.capabilities.incl(gcLocation)
    # ... etc
