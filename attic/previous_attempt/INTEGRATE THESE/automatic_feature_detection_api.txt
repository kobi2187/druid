# Automatic Feature Detection & Friction-Free Development

## Global Feature Tracking (Compile-Time)
var usedFeatures {.compileTime.}: set[AndroidFeature]
var needsContext {.compileTime.}: bool = false

template trackFeature(feature: AndroidFeature, requiresContext: bool = false) =
  static:
    usedFeatures.incl(feature)
    if requiresContext:
      needsContext = true

## Friction-Free API - Users Just Write Normal Code
proc playAudio*(buffer: AudioBuffer) =
  trackFeature(afAudio)  # Automatically tracked
  # Implementation chosen at runtime
  getAudioEngine().play(buffer)

proc showNotification*(title, message: string) =
  trackFeature(afNotifications, requiresContext = true)  # Tracks that SDK is needed
  getNotificationManager().notify(title, message)

proc getAcceleration*(): Vector3 =
  trackFeature(afSensors)
  return getSensorManager().getAcceleration()

proc enableLowLatencyAudio*() =
  trackFeature(afLowLatencyAudio)  # Implies API 26+ needed
  getAudioEngine().setLowLatency(true)

## Automatic Runtime Initialization
var androidRuntime {.global.}: AndroidRuntime

proc initAndroidRuntime*() =
  # Automatically determine what we need based on used features
  static:
    let runtimeMode = if needsContext: armHybrid else: armNdkOnly
    
  androidRuntime = AndroidRuntime(
    mode: runtimeMode,
    apiLevel: getCurrentApiLevel(),
    context: when needsContext: autoDetectContext() else: none(AndroidContext),
    usedFeatures: usedFeatures
  )
  
  # Automatic capability validation
  validateFeatures(androidRuntime)

proc validateFeatures(runtime: AndroidRuntime) =
  # Check if used features are actually available
  for feature in runtime.usedFeatures:
    if not isFeatureAvailable(runtime, feature):
      let fallback = getFallbackStrategy(feature)
      if fallback.isSome:
        echo "⚠️ ", feature, " not available, using fallback: ", fallback.get()
      else:
        echo "❌ ", feature, " not available and no fallback exists"

## Smart Context Detection (Automatic)
proc autoDetectContext(): Option[AndroidContext] =
  # Try multiple sources automatically
  result = none(AndroidContext)
  
  # 1. From GameActivity (if available)
  when compiles(getGameActivityContext()):
    if let ctx = getGameActivityContext():
      return some(ctx)
  
  # 2. From NativeActivity
  when compiles(getNativeActivityContext()):
    if let ctx = getNativeActivityContext():
      return some(ctx)
      
  # 3. From JNI global reference (if set by user)
  if globalJNIContext.isSome:
    return globalJNIContext
    
  # 4. Prompt user with helpful error
  {.error: """
  Java SDK context required but not found. 
  
  Add ONE of these to your main function:
    setAndroidContext(activity, env)           # Manual
    useGameActivity()                          # GameActivity integration  
    useNativeActivity()                        # NativeActivity integration
  
  Or disable features that need Java SDK context.
  """.}
